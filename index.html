function doGet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. Get Sweepstakes Data const sweepSheet = ss.getSheetByName("Web_Display"); const sweepData = sweepSheet.getRange(1, 1, sweepSheet.getLastRow(), 23).getValues(); // 2. Get Pro Leaderboard Data const majorSheet = ss.getSheetByName("Major_Scores"); const majorData = majorSheet ? majorSheet.getDataRange().getValues() : [["No Data"]]; // 3. Package for the website const result = { sweepstakes: convertToJSON(sweepData), major: convertToJSON(majorData) }; return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } function convertToJSON(values) { const headers = values[0]; return values.slice(1).map(row => { return headers.reduce((acc, header, i) => { acc[header] = row[i]; return acc; }, {}); }); }
